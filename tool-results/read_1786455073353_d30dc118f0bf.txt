   142→function parseJSONField(val: string | null | undefined): string[] {
   143→  if (!val) return []
   144→  try {
   145→    const parsed = JSON.parse(val)
   146→    return Array.isArray(parsed) ? parsed : []
   147→  } catch {
   148→    return []
   149→  }
   150→}
   151→
   152→function parseJSONObj(val: string | null | undefined): Record<string, unknown> {
   153→  if (!val) return {}
   154→  try {
   155→    return JSON.parse(val)
   156→  } catch {
   157→    return {}
   158→  }
   159→}

   259→        method: 'POST',
   260→        headers: { 'Content-Type': 'application/json' },
