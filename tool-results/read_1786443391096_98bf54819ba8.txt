   278→      setVerifying(true)
   279→      const res = await fetch(`/api/tasks/${taskId}/verify`, {
   280→        method: 'POST',
   281→        headers: { 'Content-Type': 'application/json' },
   282→        body: JSON.stringify({ userId }),
   283→      })
   284→      const json = await res.json()
   285→      if (!res.ok) throw new Error(json.error || 'Verification failed')
   286→      // Re-fetch task
   287→      const taskRes = await fetch(`/api/tasks/${taskId}`)
   288→      const taskJson = await taskRes.json()
   289→      if (taskRes.ok) setTask(taskJson.data)
   290→    } catch (err) {
   291→      setError(err instanceof Error ? err.message : 'Verification failed')
   292→    } finally {
   293→      setVerifying(false)
   294→    }
   295→  }
