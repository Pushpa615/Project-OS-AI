    67→    67→  { value: 'Abstract', label: 'Abstract' },
    68→    68→  { value: 'Project Report', label: 'Project Report' },

    76→    76→
    77→    77→export function AIAssistantPage({ userId, projectId, taskId }: AIAssistantPageProps) {
    78→    78→  const navigate = useNavStore((s) => s.navigate)
    79→    79→
    80→    80→  // Projects & tasks for selectors
    81→    81→  const [projects, setProjects] = useState<Project[]>([])
    82→    82→  const [tasks, setTasks] = useState<Task[]>([])
    83→    83→  const [projectsLoading, setProjectsLoading] = useState(true)
    84→    84→
    85→    85→  // Shared AI state
    86→    86→  const [loading, setLoading] = useState(false)
    87→    87→  const [error, setError] = useState<string | null>(null)
    88→    88→
    89→    89→  // Tab 1: Code Assistant
    90→    90→  const [codeProjectId, setCodeProjectId] = useState<string>(projectId || '')
    91→    91→  const [codeTaskId, setCodeTaskId] = useState<string>(taskId || '')
    92→    92→  const [codeQuestion, setCodeQuestion] = useState('')
    93→    93→  const [codeHistory, setCodeHistory] = useState<ChatMessage[]>([])
    94→    94→
    95→    95→  // Tab 2: Bug Assistant
    96→    96→  const [bugErrorMessage, setBugErrorMessage] = useState('')
    97→    97→  const [bugRelevantCode, setBugRelevantCode] = useState('')
    98→    98→  const [bugExpected, setBugExpected] = useState('')
    99→    99→  const [bugActual, setBugActual] = useState('')
   100→   100→  const [bugResponse, setBugResponse] = useState('')
   101→   101→  const [bugHistory, setBugHistory] = useState<ChatMessage[]>([])
   102→   102→
   103→   103→  // Tab 3: Documentation
   104→   104→  const [docProjectId, setDocProjectId] = useState<string>(projectId || '')
   105→   105→  const [docType, setDocType] = useState('')
   106→   106→  const [docResponse, setDocResponse] = useState('')
   107→   107→  const [copied, setCopied] = useState(false)
   108→   108→
   109→   109→  // Tab 4: Deadline Manager
   110→   110→  const [deadlineProjectId, setDeadlineProjectId] = useState<string>(projectId || '')
   111→   111→  const [deadlineResponse, setDeadlineResponse] = useState('')
   112→   112→  const [overdueTasks, setOverdueTasks] = useState<Task[]>([])
   113→   113→
   114→   114→  // Fetch projects
   115→   115→  const fetchProjects = useCallback(async () => {
   116→   116→    try {
   117→   117→      const res = await fetch(`/api/projects?userId=${userId}`)
   118→   118→      const json = await res.json()
   119→   119→      if (json.data) setProjects(json.data)
   120→   120→    } catch {
   121→   121→      // silent
   122→   122→    } finally {
   123→   123→      setProjectsLoading(false)
   124→   124→    }
   125→   125→  }, [userId])
   126→   126→
   127→   127→  // Fetch tasks for a project
   128→   128→  const fetchTasks = useCallback(async (pid: string) => {
   129→   129→    if (!pid) {
   130→   130→      setTasks([])
   131→   131→      return
   132→   132→    }
   133→   133→    try {
   134→   134→      const res = await fetch(`/api/tasks?projectId=${pid}`)
   135→   135→      const json = await res.json()
   136→   136→      if (json.data) setTasks(json.data)
   137→   137→    } catch {
   138→   138→      // silent
   139→   139→    }
   140→   140→  }, [])
   141→   141→
   142→   142→  // Fetch overdue tasks
   143→   143→  const fetchOverdue = useCallback(async (pid: string) => {
   144→   144→    if (!pid) return
   145→   145→    try {
   146→   146→      const res = await fetch(`/api/tasks?projectId=${pid}&status=overdue`)
   147→   147→      const json = await res.json()
   148→   148→      if (json.data) setOverdueTasks(json.data)
   149→   149→    } catch {
   150→   150→      // silent
   151→   151→    }
   152→   152→  }, [])
   153→   153→
   154→   154→  useEffect(() => {
   155→   155→    fetchProjects()
   156→   156→  }, [fetchProjects])
   157→   157→
   158→   158→  useEffect(() => {
   159→   159→    if (codeProjectId) fetchTasks(codeProjectId)
   160→   160→  }, [codeProjectId, fetchTasks])
   161→   161→
   162→   162→  useEffect(() => {
   163→   163→    if (deadlineProjectId) fetchOverdue(deadlineProjectId)
   164→   164→  }, [deadlineProjectId, fetchOverdue])
   165→   165→
   166→   166→  // AI request helper — accepts explicit projectId/taskId overrides
   167→   167→  const callAI = useCallback(
   168→   168→    async (type: string, question: string, context: string = '', overrides?: { projectId?: string; taskId?: string }) => {
   169→   169→      setLoading(true)
   170→   170→      setError(null)
   171→   171→      try {
   172→   172→        const res = await fetch('/api/ai/chat', {
   173→   173→          method: 'POST',
   174→   174→          headers: { 'Content-Type': 'application/json' },
   175→   175→          body: JSON.stringify({
   176→   176→            userId,
   177→   177→            projectId: overrides?.projectId || projectId || undefined,
   178→   178→            taskId: overrides?.taskId || taskId || undefined,
   179→   179→            type,
   180→   180→            question,
   181→   181→            context,
   182→   182→          }),
   183→   183→        })
   184→   184→        const json = await res.json()
   185→   185→        if (json.error) {
   186→   186→          setError(json.error)
   187→   187→          return null
   188→   188→        }
   189→   189→        return json.data.response as string
   190→   190→      } catch {
   191→   191→        setError('AI request failed. Please try again.')
   192→   192→        return null
   193→   193→      } finally {
   194→   194→        setLoading(false)
   195→   195→      }
   196→   196→    },
   197→   197→    [userId, projectId, taskId]
   198→   198→  )
   199→   199→
   200→   200→  // Code assistant submit
   201→   201→  const handleCodeSubmit = async () => {
   202→   202→    if (!codeQuestion.trim()) return
   203→   203→    const response = await callAI('code_help', codeQuestion, undefined, { projectId: codeProjectId, taskId: codeTaskId || undefined })
   204→   204→    if (response) {
   205→   205→      setCodeHistory((prev) => [
   206→   206→        { question: codeQuestion, response, type: 'code_help', timestamp: new Date().toISOString() },
   207→   207→        ...prev,
   208→   208→      ])
   209→   209→      setCodeQuestion('')
   210→   210→    }
   211→   211→  }
   212→   212→
   213→   213→  // Bug assistant submit
   214→   214→  const handleBugSubmit = async () => {
   215→   215→    if (!bugErrorMessage.trim()) return
   216→   216→    const context = [
   217→   217→      bugRelevantCode ? `Relevant Code:\n${bugRelevantCode}` : '',
   218→   218→      bugExpected ? `Expected Behavior: ${bugExpected}` : '',
   219→   219→      bugActual ? `Actual Behavior: ${bugActual}` : '',
   220→   220→    ]
   221→   221→      .filter(Boolean)
   222→   222→      .join('\n\n')
   223→   223→
   224→   224→    const response = await callAI('bug_help', bugErrorMessage, context, { projectId })
   225→   225→    if (response) {
   226→   226→      setBugHistory((prev) => [
   227→   227→        { question: bugErrorMessage, response, type: 'bug_help', timestamp: new Date().toISOString() },
   228→   228→        ...prev,
   229→   229→      ])
   230→   230→      setBugResponse(response)
   231→   231→      setBugErrorMessage('')
   232→   232→      setBugRelevantCode('')
   233→   233→      setBugExpected('')
   234→   234→      setBugActual('')
   235→   235→    }
   236→   236→  }
   237→   237→
   238→   238→  // Documentation submit
   239→   239→  const handleDocGenerate = async () => {
   240→   240→    if (!docProjectId || !docType) return
   241→   241→    const question = `Generate a ${docType} document for this project. Include all relevant sections, be comprehensive and professional.`
   242→   242→    const response = await callAI('general', question, undefined, { projectId: docProjectId })
   243→   243→    if (response) {
   244→   244→      setDocResponse(response)
   245→   245→    }
   246→   246→  }
   247→   247→
   248→   248→  // Deadline manager submit
   249→   249→  const handleDeadlineAnalyze = async () => {
   250→   250→    if (!deadlineProjectId) return
   251→   251→    const overdueInfo = overdueTasks
   252→   252→      .map((t) => ` - ${t.title} (${t.status})`)
   253→   253→      .join('\n')
   254→   254→    const question = `Analyze the current project timeline and provide: 1) Timeline status assessment, 2) Overdue task analysis, 3) Risk identification, 4) Specific AI-powered recommendations for getting back on track.\n\nOverdue tasks:\n${overdueInfo || 'None'}`
   255→   255→    const response = await callAI('general', question, undefined, { projectId: deadlineProjectId })
   256→   256→    if (response) {
   257→   257→      setDeadlineResponse(response)
   258→   258→    }
   259→   259→  }
   260→   260→
   261→   261→  // Copy to clipboard
   262→   262→  const handleCopy = (text: string) => {
   263→   263→    navigator.clipboard.writeText(text)
   264→   264→    setCopied(true)
   265→   265→    setTimeout(() => setCopied(false), 2000)
   266→   266→  }
   267→   267→
   268→   268→  const clearHistory = (setter: React.Dispatch<React.SetStateAction<ChatMessage[]>>) => {
   269→   269→    setter([])
   270→   270→  }
   271→   271→
   272→   272→  return (
   273→   273→    <div className="min-h-screen bg-background">
   274→   274→      <div className="max-w-4xl mx-auto p-4 sm:p-6 space-y-6">
   275→   275→        {/* Header */}
   276→   276→        <div className="flex items-center gap-3">
   277→   277→          <div className="h-10 w-10 rounded-xl bg-primary/10 flex items-center justify-center">
   278→   278→            <Bot className="h-5 w-5 text-primary" />
   279→   279→          </div>
   280→   280→          <div>
   281→   281→            <h1 className="text-2xl font-bold tracking-tight">AI Assistant</h1>
   282→   282→            <p className="text-sm text-muted-foreground">
   283→   283→              Powered by AI to help you build better
   284→   284→            </p>
   285→   285→          </div>
   286→   286→        </div>
   287→   287→
   288→   288→        {/* Tabs */}
   289→   289→        <Tabs defaultValue="code" className="space-y-4">
   290→   290→          <TabsList className="grid w-full grid-cols-2 sm:grid-cols-4">
   291→   291→            <TabsTrigger value="code" className="gap-1.5 text-xs sm:text-sm">
   292→   292→              <Code2 className="h-3.5 w-3.5" />
   293→   293→              <span className="hidden sm:inline">Code</span>
   294→   294→            </TabsTrigger>
   295→   295→            <TabsTrigger value="bug" className="gap-1.5 text-xs sm:text-sm">
   296→   296→              <Bug className="h-3.5 w-3.5" />
   297→   297→              <span className="hidden sm:inline">Bug</span>
   298→   298→            </TabsTrigger>
   299→   299→            <TabsTrigger value="docs" className="gap-1.5 text-xs sm:text-sm">
   300→   300→              <FileText className="h-3.5 w-3.5" />
   301→   301→              <span className="hidden sm:inline">Docs</span>
   302→   302→            </TabsTrigger>
   303→   303→            <TabsTrigger value="deadline" className="gap-1.5 text-xs sm:text-sm">
   304→   304→              <CalendarClock className="h-3.5 w-3.5" />
   305→   305→              <span className="hidden sm:inline">Deadlines</span>
   306→   306→            </TabsTrigger>
   307→   307→          </TabsList>
   308→   308→
   309→   309→          {/* =========== TAB 1: Code Assistant =========== */}
   310→   310→          <TabsContent value="code" className="space-y-4">
   311→   311→            <Card>
   312→   312→              <CardHeader className="pb-3">
   313→   313→                <CardTitle className="text-base font-semibold flex items-center gap-2">
   314→   314→                  <Code2 className="h-4 w-4 text-emerald-500" />
   315→   315→                  Code Assistant
   316→   316→                </CardTitle>
   317→   317→              </CardHeader>
   318→   318→              <CardContent className="space-y-4">
   319→   319→                {/* Project selector */}
   320→   320→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   321→   321→                  <div className="space-y-1.5">
   322→   322→                    <label className="text-xs font-medium text-muted-foreground">Project</label>
   323→   323→                    <Select value={codeProjectId} onValueChange={(v) => setCodeProjectId(v)}>
   324→   324→                      <SelectTrigger className="h-9">
   325→   325→                        <SelectValue placeholder="Select project" />
   326→   326→                      </SelectTrigger>
   327→   327→                      <SelectContent>
   328→   328→                        {projects.map((p) => (
   329→   329→                          <SelectItem key={p.id} value={p.id}>
   330→   330→                            {p.name}
   331→   331→                          </SelectItem>
   332→   332→                        ))}
   333→   333→                      </SelectContent>
   334→   334→                    </Select>
   335→   335→                  </div>
   336→   336→                  {codeProjectId && (
   337→   337→                    <div className="space-y-1.5">
   338→   338→                      <label className="text-xs font-medium text-muted-foreground">Task (optional)</label>
   339→   339→                      <Select value={codeTaskId} onValueChange={(v) => setCodeTaskId(v)}>
   340→   340→                        <SelectTrigger className="h-9">
   341→   341→                          <SelectValue placeholder="Select task" />
   342→   342→                        </SelectTrigger>
   343→   343→                        <SelectContent>
   344→   344→                          {tasks.map((t) => (
   345→   345→                            <SelectItem key={t.id} value={t.id}>
   346→   346→                              {t.title}
   347→   347→                            </SelectItem>
   348→   348→                          ))}
   349→   349→                        </SelectContent>
   350→   350→                      </Select>
   351→   351→                    </div>
   352→   352→                  )}
   353→   353→                </div>
   354→   354→
   355→   355→                <div className="space-y-1.5">
   356→   356→                  <label className="text-xs font-medium text-muted-foreground">Ask for help</label>
   357→   357→                  <Textarea
   358→   358→                    placeholder="Describe what you need help with..."
   359→   359→                    value={codeQuestion}
   360→   360→                    onChange={(e) => setCodeQuestion(e.target.value)}
   361→   361→                    rows={3}
   362→   362→                    onKeyDown={(e) => {
   363→   363→                      if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) handleCodeSubmit()
   364→   364→                    }}
   365→   365→                  />
   366→   366→                </div>
   367→   367→
   368→   368→                {error && (
   369→   369→                  <p className="text-sm text-destructive flex items-center gap-1">
   370→   370→                    <AlertTriangle className="h-3 w-3" /> {error}
   371→   371→                  </p>
   372→   372→                )}
   373→   373→
   374→   374→                <Button onClick={handleCodeSubmit} disabled={loading || !codeQuestion.trim()}>
   375→   375→                  {loading ? (
   376→   376→                    <>
   377→   377→                      <Loader2 className="h-4 w-4 animate-spin mr-2" />
   378→   378→                      Thinking...
   379→   379→                    </>
   380→   380→                  ) : (
   381→   381→                    <>
   382→   382→                      <Send className="h-4 w-4 mr-2" />
   383→   383→                      Get Help
   384→   384→                    </>
   385→   385→                  )}
   386→   386→                </Button>
   387→   387→              </CardContent>
   388→   388→            </Card>
   389→   389→
   390→   390→            {/* Code history */}
   391→   391→            {codeHistory.length > 0 && (
   392→   392→              <div className="space-y-3">
   393→   393→                <div className="flex items-center justify-between">
   394→   394→                  <h3 className="text-sm font-semibold text-muted-foreground">History</h3>
   395→   395→                  <Button
   396→   396→                    variant="ghost"
   397→   397→                    size="sm"
   398→   398→                    className="text-xs h-7"
   399→   399→                    onClick={() => clearHistory(setCodeHistory)}
   400→   400→                  >
   401→   401→                    <Trash2 className="h-3 w-3 mr-1" /> Clear
   402→   402→                  </Button>
   403→   403→                </div>
   404→   404→                {codeHistory.map((msg, i) => (
   405→   405→                  <Card key={i} className="overflow-hidden">
   406→   406→                    <CardContent className="p-4 space-y-3">
   407→   407→                      <div className="flex items-start gap-2">
   408→   408→                        <div className="h-7 w-7 rounded-full bg-primary/10 flex items-center justify-center shrink-0 mt-0.5">
   409→   409→                          <span className="text-xs font-medium text-primary">You</span>
   410→   410→                        </div>
   411→   411→                        <div className="flex-1 min-w-0">
   412→   412→                          <p className="text-sm font-medium">{msg.question}</p>
   413→   413→                          <p className="text-[10px] text-muted-foreground mt-0.5">
   414→   414→                            {formatDistanceToNow(parseISO(msg.timestamp), { addSuffix: true })}
   415→   415→                          </p>
   416→   416→                        </div>
   417→   417→                      </div>
   418→   418→                      <Separator />
   419→   419→                      <div className="flex items-start gap-2">
   420→   420→                        <div className="h-7 w-7 rounded-full bg-emerald-100 dark:bg-emerald-900/20 flex items-center justify-center shrink-0 mt-0.5">
   421→   421→                          <Sparkles className="h-3.5 w-3.5 text-emerald-600 dark:text-emerald-400" />
   422→   422→                        </div>
   423→   423→                        <div className="flex-1 min-w-0">
   424→   424→                          <p className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">
   425→   425→                            {msg.response}
   426→   426→                          </p>
   427→   427→                        </div>
   428→   428→                      </div>
   429→   429→                    </CardContent>
   430→   430→                  </Card>
   431→   431→                ))}
   432→   432→              </div>
   433→   433→            )}
   434→   434→          </TabsContent>
   435→   435→
   436→   436→          {/* =========== TAB 2: Bug Assistant =========== */}
   437→   437→          <TabsContent value="bug" className="space-y-4">
   438→   438→            <Card>
   439→   439→              <CardHeader className="pb-3">
   440→   440→                <CardTitle className="text-base font-semibold flex items-center gap-2">
   441→   441→                  <Bug className="h-4 w-4 text-red-500" />
   442→   442→                  Bug Assistant
   443→   443→                </CardTitle>
   444→   444→              </CardHeader>
   445→   445→              <CardContent className="space-y-4">
   446→   446→                <div className="space-y-1.5">
   447→   447→                  <label className="text-xs font-medium text-muted-foreground">Error Message</label>
   448→   448→                  <Textarea
   449→   449→                    placeholder="Paste the error message or stack trace..."
   450→   450→                    value={bugErrorMessage}
   451→   451→                    onChange={(e) => setBugErrorMessage(e.target.value)}
   452→   452→                    rows={3}
   453→   453→                  />
   454→   454→                </div>
   455→   455→
   456→   456→                <div className="space-y-1.5">
   457→   457→                  <label className="text-xs font-medium text-muted-foreground">
   458→   458→                    Relevant Code (optional)
   459→   459→                  </label>
   460→   460→                  <Textarea
   461→   461→                    placeholder="Paste the code where the bug occurs..."
   462→   462→                    value={bugRelevantCode}
   463→   463→                    onChange={(e) => setBugRelevantCode(e.target.value)}
   464→   464→                    rows={4}
   465→   465→                    className="font-mono text-xs"
   466→   466→                  />
   467→   467→                </div>
   468→   468→
   469→   469→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   470→   470→                  <div className="space-y-1.5">
   471→   471→                    <label className="text-xs font-medium text-muted-foreground">Expected Behavior</label>
   472→   472→                    <Input
