   147→function parseJSONField(val: string | null | undefined): string[] {
   148→  if (!val) return []
   149→  try {
   150→    const parsed = JSON.parse(val)
   151→    return Array.isArray(parsed) ? parsed : []
   152→  } catch {
   153→    return []
   154→  }
   155→}
   156→
   157→function parseJSONObj(val: string | null | undefined): Record<string, unknown> {
   158→  if (!val) return {}
   159→  try {
   160→    return JSON.parse(val)
   161→  } catch {
   162→    return {}
   163→  }
   164→}

   332→    const file = e.target.files?.[0]

   335→      setUploadingFile(true)

   337→        method: 'POST',

   339→        body: JSON.stringify({
   340→          projectId,
   341→          userId,
   342→          fileName: file.name,
   343→          fileSize: file.size,
   344→          mimeType: file.type,
   345→        }),
   346→      })
   347→      if (!res.ok) throw new Error('Failed to upload file')
   348→      await fetchFiles()
   349→      const projRes = await fetch(`/api/projects/${projectId}`)
   350→      const projJson = await projRes.json()

   352→    } catch { /* ignore */ } finally {
   353→      setUploadingFile(false)
   354→      e.target.value = ''
