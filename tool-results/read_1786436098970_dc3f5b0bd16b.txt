   140→function formatDate(dateStr: string | null): string {
   141→  if (!dateStr) return 'Not set'
   142→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
   143→}

   147→export function TaskDetailPage({ taskId, userId }: { taskId: string; userId: string }) {
   148→  const navigate = useNavStore((s) => s.navigate)
   149→  const [task, setTask] = useState<TaskDetail | null>(null)
   150→  const [loading, setLoading] = useState(true)
   151→  const [error, setError] = useState<string | null>(null)
   152→  const [activeTab, setActiveTab] = useState('details')
   153→  const [saving, setSaving] = useState(false)
   154→  const [saveError, setSaveError] = useState<string | null>(null)
   155→  const [saveSuccess, setSaveSuccess] = useState(false)
   156→  const [comments, setComments] = useState<Comment[]>([])
   157→  const [commentInput, setCommentInput] = useState('')
   158→  const [submittingComment, setSubmittingComment] = useState(false)
   159→
   160→  // AI Help state
   161→  const [aiMessages, setAiMessages] = useState<{ role: 'user' | 'assistant'; content: string }[]>([])
   162→  const [aiQuestion, setAiQuestion] = useState('')
   163→  const [aiLoading, setAiLoading] = useState(false)
   164→
   165→  // Evidence dialog state
   166→  const [evidenceOpen, setEvidenceOpen] = useState(false)
   167→  const [evidenceType, setEvidenceType] = useState('link')
   168→  const [evidenceTitle, setEvidenceTitle] = useState('')
   169→  const [evidenceDesc, setEvidenceDesc] = useState('')
   170→  const [evidenceUrl, setEvidenceUrl] = useState('')
   171→  const [evidenceSaving, setEvidenceSaving] = useState(false)
   172→
   173→  // Verifying state
   174→  const [verifying, setVerifying] = useState(false)
   175→
   176→  // Editable fields
   177→  const [editStatus, setEditStatus] = useState('')
   178→  const [editCompletion, setEditCompletion] = useState(0)
   179→  const [editHours, setEditHours] = useState('')
   180→  const [editEstimated, setEditEstimated] = useState('')
   181→  const [depsExpanded, setDepsExpanded] = useState(true)
   182→  const [depOfExpanded, setDepOfExpanded] = useState(true)
   183→
   184→  // Fetch task
   185→  useEffect(() => {
   186→    async function fetchTask() {
   187→      try {
   188→        setLoading(true)
   189→        const res = await fetch(`/api/tasks/${taskId}`)
   190→        const json = await res.json()
   191→        if (!res.ok) throw new Error(json.error || 'Failed to fetch task')
   192→        const t = json.data
   193→        setTask(t)
   194→        setEditStatus(t.status)
   195→        setEditCompletion(t.completionPercent)
   196→        setEditHours(t.actualHours?.toString() || '')
   197→        setEditEstimated(t.estimatedHours?.toString() || '')
   198→        setComments(t.taskComments || [])
   199→      } catch (err) {
   200→        setError(err instanceof Error ? err.message : 'Something went wrong')
   201→      } finally {
   202→        setLoading(false)
   203→      }
   204→    }
   205→    fetchTask()
   206→  }, [taskId])
   207→
   208→  // Save task details
   209→  async function handleSave() {
   210→    try {
   211→      setSaving(true)
   212→      setSaveError(null)
   213→      setSaveSuccess(false)
   214→      const body: Record<string, unknown> = {
   215→        userId,
   216→        status: editStatus,
   217→        completionPercent: editCompletion,
   218→        actualHours: editHours ? parseFloat(editHours) : null,
   219→        estimatedHours: editEstimated ? parseFloat(editEstimated) : null,
   220→      }
   221→      const res = await fetch(`/api/tasks/${taskId}`, {
   222→        method: 'PUT',
   223→        headers: { 'Content-Type': 'application/json' },
   224→        body: JSON.stringify(body),
   225→      })
   226→      const json = await res.json()
   227→      if (!res.ok) throw new Error(json.error || 'Failed to update task')
   228→      setSaveSuccess(true)
   229→      setTimeout(() => setSaveSuccess(false), 3000)
   230→      // Re-fetch
   231→      const taskRes = await fetch(`/api/tasks/${taskId}`)
   232→      const taskJson = await taskRes.json()
   233→      if (taskRes.ok) setTask(taskJson.data)
   234→    } catch (err) {
   235→      setSaveError(err instanceof Error ? err.message : 'Failed to save')
   236→    } finally {
   237→      setSaving(false)
   238→    }
   239→  }
   240→
   241→  // Add evidence
   242→  async function handleAddEvidence() {
   243→    try {
   244→      setEvidenceSaving(true)
   245→      const body = {
   246→        type: evidenceType,
   247→        title: evidenceTitle,
   248→        description: evidenceDesc,
   249→        url: evidenceUrl,
   250→      }
   251→      const res = await fetch(`/api/tasks/${taskId}/evidence`, {
   252→        method: 'POST',
   253→        headers: { 'Content-Type': 'application/json' },
   254→        body: JSON.stringify(body),
   255→      })
   256→      const json = await res.json()
   257→      if (!res.ok) throw new Error(json.error || 'Failed to add evidence')
   258→      setEvidenceOpen(false)
   259→      setEvidenceTitle('')
   260→      setEvidenceDesc('')
   261→      setEvidenceUrl('')
   262→      setEvidenceType('link')
   263→      // Re-fetch
   264→      const taskRes = await fetch(`/api/tasks/${taskId}`)
   265→      const taskJson = await taskRes.json()
   266→      if (taskRes.ok) setTask(taskJson.data)
   267→    } catch (err) {
   268→      setError(err instanceof Error ? err.message : 'Failed to add evidence')
   269→    } finally {
   270→      setEvidenceSaving(false)
   271→    }
   272→  }
   273→
   274→  // Verify with AI
   275→  async function handleVerify() {
   276→    try {
   277→      setVerifying(true)
   278→      const res = await fetch(`/api/tasks/${taskId}/verify`, {
   279→        method: 'POST',
   280→        headers: { 'Content-Type': 'application/json' },
   281→        body: JSON.stringify({ userId }),
   282→      })
   283→      const json = await res.json()
   284→      if (!res.ok) throw new Error(json.error || 'Verification failed')
   285→      // Re-fetch task
   286→      const taskRes = await fetch(`/api/tasks/${taskId}`)
   287→      const taskJson = await taskRes.json()
   288→      if (taskRes.ok) setTask(taskJson.data)
   289→    } catch (err) {
   290→      setError(err instanceof Error ? err.message : 'Verification failed')
   291→    } finally {
   292→      setVerifying(false)
   293→    }
   294→  }
   295→
   296→  // Add comment
   297→  async function handleAddComment() {
   298→    if (!commentInput.trim()) return
   299→    try {
   300→      setSubmittingComment(true)
   301→      const res = await fetch('/api/comments', {
   302→        method: 'POST',
   303→        headers: { 'Content-Type': 'application/json' },
   304→        body: JSON.stringify({
   305→          userId,
   306→          taskId,
   307→          content: commentInput.trim(),
   308→        }),
   309→      })
   310→      const json = await res.json()
   311→      if (!res.ok) throw new Error(json.error || 'Failed to add comment')
   312→      setCommentInput('')
   313→      const newComment = json.data
   314→      setComments((prev) => [...prev, newComment])
   315→    } catch (err) {
   316→      setError(err instanceof Error ? err.message : 'Failed to add comment')
   317→    } finally {
   318→      setSubmittingComment(false)
   319→    }
   320→  }
   321→
   322→  // AI Chat
   323→  async function handleAiChat() {
   324→    if (!aiQuestion.trim()) return
   325→    const question = aiQuestion.trim()
   326→    setAiQuestion('')
   327→    setAiMessages((prev) => [...prev, { role: 'user', content: question }])
   328→    setAiLoading(true)
   329→    try {
   330→      const res = await fetch('/api/ai/chat', {
   331→        method: 'POST',
   332→        headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({
   334→          userId,
   335→          taskId,
   336→          projectId: task?.project.id,
   337→          type: 'code_help',
   338→          question,
   339→        }),
   340→      })
   341→      const json = await res.json()
   342→      if (!res.ok) throw new Error(json.error || 'AI chat failed')
   343→      setAiMessages((prev) => [...prev, { role: 'assistant', content: json.data?.response || 'No response' }])
   344→    } catch (err) {
   345→      setAiMessages((prev) => [...prev, { role: 'assistant', content: 'Sorry, an error occurred. Please try again.' }])
   346→    } finally {
   347→      setAiLoading(false)
   348→    }
   349→  }
   350→
   351→  // ==================== LOADING STATE ====================
   352→  if (loading) {
   353→    return (
   354→      <div className="min-h-screen bg-background">
   355→        <div className="border-b bg-background">
   356→          <div className="max-w-4xl mx-auto px-4 sm:px-6 py-4 flex items-center gap-3">
   357→            <Skeleton className="h-8 w-8 rounded-md" />
   358→            <Skeleton className="h-6 w-64" />
   359→          </div>
   360→        </div>
   361→        <div className="max-w-4xl mx-auto px-4 sm:px-6 py-6 space-y-4">
   362→          {Array.from({ length: 6 }).map((_, i) => (
   363→            <Skeleton key={i} className="h-16 w-full rounded-lg" />

   365→        </div>
   366→      </div>
   367→    )

   376→            <ArrowLeft className="h-4 w-4" />
   377→            Back to Tasks
   378→          </Button>
   379→        </div>
   380→      </div>
   381→    )
   382→  }
   383→

   385→

   387→

   389→    <div className="min-h-screen bg-background">
   390→      {/* Header */}
   391→      <div className="border-b bg-background sticky top-0 z-40">

   394→            <Button variant="ghost" size="icon" onClick={() => task.project ? navigate('project-detail', { id: task.project.id }) : navigate('tasks')}>

   398→              <div className="flex items-center gap-2 flex-wrap">

   401→                  {task.status.replace(/_/g, ' ')}
   402→                </Badge>
   403→                <Badge variant="outline" className={`text-[10px] px-2 py-0 ${pConfig.color}`}>
   404→                  {pConfig.label}
   405→                </Badge>
   406→              </div>
   407→            </div>
   408→          </div>
   409→        </div>
   410→      </div>
   411→

   413→      <div className="max-w-4xl mx-auto px-4 sm:px-6 py-6">
   414→        {error && (
   415→          <div className="mb-4 p-3 rounded-lg bg-destructive/10 border border-destructive/20 text-destructive text-sm">
   416→            {error}
   417→          </div>
   418→        )}
   419→
   420→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   421→          <TabsList className="w-full flex overflow-x-auto mb-6 bg-muted p-1 rounded-lg h-auto gap-1">
   422→            <TabsTrigger value="details" className="gap-1.5 text-xs sm:text-sm flex-1 shrink-0">
   423→              <FileText className="h-3.5 w-3.5 hidden sm:block" />
   424→              Details
   425→            </TabsTrigger>
   426→            <TabsTrigger value="evidence" className="gap-1.5 text-xs sm:text-sm flex-1 shrink-0">
   427→              <ImageIcon className="h-3.5 w-3.5 hidden sm:block" />

   455→                    {task.description}
