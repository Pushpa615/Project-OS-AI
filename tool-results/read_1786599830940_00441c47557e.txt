    74→function ProgressRing({ value, size = 44, strokeWidth = 4, className = '' }: { value: number; size?: number; strokeWidth?: number; className?: string }) {
    75→  const radius = (size - strokeWidth) / 2
    76→  const circumference = radius * 2 * Math.PI
    77→  const offset = circumference - (value / 100) * circumference
    78→  const color = value >= 75 ? 'text-emerald-500' : value >= 40 ? 'text-amber-500' : 'text-red-400'
    79→
    80→  return (
    81→    <div className={`relative inline-flex items-center justify-center ${className}`}>
    82→      <svg width={size} height={size} className="-rotate-90">
    83→        <circle
    84→          cx={size / 2}
    85→          cy={size / 2}
    86→          r={radius}
    87→          fill="none"
    88→          stroke="currentColor"
    89→          strokeWidth={strokeWidth}
    90→          className="text-muted/30"
    91→        />
    92→        <circle
    93→          cx={size / 2}
    94→          cy={size / 2}
    95→          r={radius}
    96→          fill="none"
    97→          stroke="currentColor"
    98→          strokeWidth={strokeWidth}
    99→          strokeDasharray={circumference}
   100→          strokeDashoffset={offset}
   101→          strokeLinecap="round"
   102→          className={`${color} transition-all duration-700 ease-out`}
   103→        />
   104→      </svg>
   105→      <span className="absolute text-[10px] font-bold text-foreground">{Math.round(value)}%</span>
   106→    </div>
   107→  )
   108→}

   112→function formatDate(dateStr: string | null): string {
   113→  if (!dateStr) return 'No deadline'
   114→  try {
   115→    return new Date(dateStr).toLocaleDateString('en-US', {
   116→      month: 'short',
   117→      day: 'numeric',
   118→      year: 'numeric',
   119→    })
   120→  } catch {
   121→    return 'Invalid date'
   122→  }
   123→}
   124→
   125→function formatRelativeTime(dateStr: string): string {
   126→  try {
   127→    const date = new Date(dateStr)
   128→    const now = new Date()
   129→    const diffMs = now.getTime() - date.getTime()
   130→    const diffMins = Math.floor(diffMs / 60000)
   131→    if (diffMins < 1) return 'Just now'
   132→    if (diffMins < 60) return `${diffMins}m ago`
   133→    const diffHours = Math.floor(diffMins / 60)
   134→    if (diffHours < 24) return `${diffHours}h ago`
   135→    const diffDays = Math.floor(diffHours / 24)
   136→    if (diffDays < 7) return `${diffDays}d ago`
   137→    return formatDate(dateStr)
   138→  } catch {
   139→    return ''
   140→  }
   141→}

   185→function getDaysUntil(dateStr: string | null): number | null {
   186→  if (!dateStr) return null
   187→  const diff = new Date(dateStr).getTime() - new Date().getTime()
   188→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   189→}
