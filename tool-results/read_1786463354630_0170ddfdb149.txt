    69→    69→function formatDate(dateStr: string | null): string {
    70→    70→  if (!dateStr) return 'No deadline'
    71→    71→  try {
    72→    72→    return new Date(dateStr).toLocaleDateString('en-US', {
    73→    73→      month: 'short',
    74→    74→      day: 'numeric',
    75→    75→      year: 'numeric',
    76→    76→    })
    77→    77→  } catch {
    78→    78→    return 'Invalid date'
    79→    79→  }
    80→    80→}
    81→    81→
    82→    82→function formatRelativeTime(dateStr: string): string {
    83→    83→  try {
    84→    84→    const date = new Date(dateStr)
    85→    85→    const now = new Date()
    86→    86→    const diffMs = now.getTime() - date.getTime()
    87→    87→    const diffMins = Math.floor(diffMs / 60000)
    88→    88→    if (diffMins < 1) return 'Just now'
    89→    89→    if (diffMins < 60) return `${diffMins}m ago`
    90→    90→    const diffHours = Math.floor(diffMins / 60)
    91→    91→    if (diffHours < 24) return `${diffHours}h ago`
    92→    92→    const diffDays = Math.floor(diffHours / 24)
    93→    93→    if (diffDays < 7) return `${diffDays}d ago`
    94→    94→    return formatDate(dateStr)
    95→    95→  } catch {
    96→    96→    return ''
    97→    97→  }
    98→    98→}
