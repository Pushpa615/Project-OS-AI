   125→   125→  const form = useForm<z.infer<typeof loginSchema>>({

   130→   130→  async function onSubmit(values: z.infer<typeof loginSchema>) {

   134→   134→      const result = await signIn('credentials', {

   145→   145→      const sessionRes = await fetch('/api/auth/session')

   149→   149→        const userId = (session.user as Record<string, string>).id

   152→   152→            id: userId,
   153→   153→            email: session.user.email,
   154→   154→            name: session.user.name || '',

   260→   260→function SignupForm({ onSuccess, onSwitch }: { onSuccess: AuthPagesProps['onAuthSuccess']; onSwitch: (v: AuthView) => void }) {
   261→   261→  const [showPassword, setShowPassword] = useState(false)
   262→   262→  const [showConfirmPassword, setShowConfirmPassword] = useState(false)
   263→   263→  const [serverError, setServerError] = useState('')
   264→   264→  const [loading, setLoading] = useState(false)
   265→   265→
   266→   266→  const form = useForm<z.infer<typeof signupSchema>>({
   267→   267→    resolver: zodResolver(signupSchema),
   268→   268→    defaultValues: { name: '', email: '', password: '', confirmPassword: '' },
   269→   269→  })
   270→   270→
   271→   271→  async function onSubmit(values: z.infer<typeof signupSchema>) {
   272→   272→    setServerError('')
   273→   273→    setLoading(true)
   274→   274→    try {
   275→   275→      const res = await fetch('/api/auth/register', {
   276→   276→        method: 'POST',
   277→   277→        headers: { 'Content-Type': 'application/json' },
   278→   278→        body: JSON.stringify({ name: values.name, email: values.email, password: values.password }),
   279→   279→      })
   280→   280→
   281→   281→      const regData = await res.json()
   282→   282→
   283→   283→      if (!res.ok) {
   284→   284→        setServerError(regData.error || 'Registration failed. Please try again.')
   285→   285→        return
   286→   286→      }
   287→   287→
   288→   288→      const registeredUser = regData.data
   289→   289→
   290→   290→      // Sign in to establish session cookie
   291→   291→      const result = await signIn('credentials', {
   292→   292→        email: values.email,
   293→   293→        password: values.password,
   294→   294→        redirect: false,
   295→   295→      })
   296→   296→
   297→   297→      if (!result?.ok) {
   298→   298→        setServerError('Registration succeeded but login failed. Please try signing in manually.')
   299→   299→        return
   300→   300→      }
   301→   301→
   302→   302→      // Small delay to ensure session cookie is set
   303→   303→      await new Promise((r) => setTimeout(r, 300))
   304→   304→
   305→   305→      const sessionRes = await fetch('/api/auth/session')
   306→   306→      const session = await sessionRes.json()
   307→   307→      const sessionUser = session?.user as Record<string, string> | undefined
   308→   308→
   309→   309→      if (sessionUser?.id) {
   310→   310→        onSuccess({
   311→   311→          id: sessionUser.id,
   312→   312→          email: sessionUser.email || values.email,
   313→   313→          name: sessionUser.name || values.name,
   314→   314→          role: sessionUser.role || 'member',
   315→   315→        })
   316→   316→      } else if (registeredUser) {
   317→   317→        // Fallback: use registered user data if session fetch failed
   318→   318→        onSuccess({
   319→   319→          id: registeredUser.id,
   320→   320→          email: registeredUser.email,
   321→   321→          name: registeredUser.name || values.name,
   322→   322→          role: registeredUser.role || 'member',
   323→   323→        })
   324→   324→      }
   325→   325→    } catch {
   326→   326→      setServerError('An unexpected error occurred. Please try again.')
   327→   327→    } finally {
   328→   328→      setLoading(false)
   329→   329→    }
   330→   330→  }
   331→   331→
   332→   332→  return (
   333→   333→    <div className="space-y-5">
   334→   334→      <div className="space-y-1.5">
   335→   335→        <h2 className="text-2xl font-bold tracking-tight">Create your account</h2>
   336→   336→        <p className="text-sm text-muted-foreground">Start managing projects with AI today</p>
   337→   337→      </div>
   338→   338→
   339→   339→      {serverError && (
   340→   340→        <div className="rounded-lg border border-destructive/50 bg-destructive/10 px-4 py-3 text-sm text-destructive">
   341→   341→          {serverError}
   342→   342→        </div>
   343→   343→      )}
   344→   344→
   345→   345→      <Form {...form}>
   346→   346→        <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   347→   347→          <FormField
   348→   348→            control={form.control}
   349→   349→            name="name"
   350→   350→            render={({ field }) => (
   351→   351→              <FormItem>
   352→   352→                <FormLabel>Full Name</FormLabel>
   353→   353→                <FormControl>
   354→   354→                  <div className="relative">

   558→   558→function ResetPasswordForm({ onSwitch }: { onSwitch: (v: AuthView) => void }) {

   564→   564→  const form = useForm<z.infer<typeof resetPasswordSchema>>({
   565→   565→    resolver: zodResolver(resetPasswordSchema),

   569→   569→  async function onSubmit(values: z.infer<typeof resetPasswordSchema>) {

   574→   574→        method: 'POST',
