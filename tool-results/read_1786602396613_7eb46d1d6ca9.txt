   118→// ==================== LOGIN FORM ====================

   120→function LoginForm({ onSuccess, onSwitch }: { onSuccess: AuthPagesProps['onAuthSuccess']; onSwitch: (v: AuthView) => void }) {
   121→  const [showPassword, setShowPassword] = useState(false)
   122→  const [serverError, setServerError] = useState('')
   123→  const [loading, setLoading] = useState(false)
   124→
   125→  const form = useForm<z.infer<typeof loginSchema>>({
   126→    resolver: zodResolver(loginSchema),
   127→    defaultValues: { email: '', password: '' },
   128→  })
   129→
   130→  async function onSubmit(values: z.infer<typeof loginSchema>) {
   131→    setServerError('')
   132→    setLoading(true)
   133→    try {
   134→      const result = await signIn('credentials', {
   135→        email: values.email,
   136→        password: values.password,
   137→        redirect: false,
   138→      })
   139→
   140→      if (result?.error) {
   141→        setServerError('Invalid email or password. Please try again.')
   142→        return
   143→      }
   144→
   145→      const sessionRes = await fetch('/api/auth/session')
   146→      const session = await sessionRes.json()
   147→
   148→      if (session?.user) {
   149→        const userId = (session.user as Record<string, string>).id
   150→        if (userId) {
   151→          onSuccess({
   152→            id: userId,
   153→            email: session.user.email,
   154→            name: session.user.name || '',
   155→            role: (session.user as Record<string, string>).role || 'member',
   156→          })
   157→        } else {
   158→          const profileRes = await fetch(`/api/auth/register?email=${encodeURIComponent(values.email)}`)
   159→          const profileData = await profileRes.json()
   160→          if (profileData.data) {
   161→            onSuccess({
   162→              id: profileData.data.id,
   163→              email: profileData.data.email,
   164→              name: profileData.data.name || '',
   165→              role: profileData.data.role || 'member',
   166→            })
   167→          }
   168→        }
   169→      }
   170→    } catch {
   171→      setServerError('An unexpected error occurred. Please try again.')
   172→    } finally {
   173→      setLoading(false)
   174→    }
   175→  }
   176→
   177→  return (
   178→    <div className="space-y-5">
   179→      <div className="space-y-1.5">
   180→        <h2 className="text-2xl font-bold tracking-tight">Welcome back</h2>
   181→        <p className="text-sm text-muted-foreground">Sign in to continue to Project OS AI</p>
   182→      </div>
   183→
   184→      {serverError && (
   185→        <div className="rounded-lg border border-destructive/50 bg-destructive/10 px-4 py-3 text-sm text-destructive">
   186→          {serverError}
   187→        </div>
   188→      )}
   189→
   190→      <Form {...form}>
   191→        <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   192→          <FormField
   193→            control={form.control}
   194→            name="email"
   195→            render={({ field }) => (
   196→              <FormItem>
   197→                <FormLabel>Email</FormLabel>
   198→                <FormControl>
   199→                  <div className="relative">
   200→                    <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   201→                    <Input placeholder="you@example.com" className="pl-10" {...field} />
   202→                  </div>
   203→                </FormControl>
   204→                <FormMessage />
   205→              </FormItem>
   206→            )}
   207→          />
   208→          <FormField
   209→            control={form.control}
   210→            name="password"
   211→            render={({ field }) => (
   212→              <FormItem>
   213→                <div className="flex items-center justify-between">
   214→                  <FormLabel>Password</FormLabel>
   215→                  <Button type="button" variant="link" className="px-0 h-auto text-xs text-muted-foreground hover:text-primary" onClick={() => onSwitch('forgot-password')}>
   216→                    Forgot password?
   217→                  </Button>
   218→                </div>
   219→                <FormControl>
   220→                  <div className="relative">
   221→                    <Lock className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   222→                    <Input type={showPassword ? 'text' : 'password'} placeholder="Enter your password" className="pl-10 pr-10" {...field} />
   223→                    <button
   224→                      type="button"
   225→                      onClick={() => setShowPassword(!showPassword)}
   226→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition-colors"
   227→                    >
   228→                      {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}

   232→                <FormMessage />
   233→              </FormItem>
   234→            )}
   235→          />
   236→          <Button type="submit" className="w-full" disabled={loading}>
   237→            {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   238→            Sign In
   239→          </Button>
   240→        </form>
   241→      </Form>

   243→      <div className="relative">
