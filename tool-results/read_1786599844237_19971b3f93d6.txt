    48→// ==================== TYPES ====================
    49→
    50→interface ProfileData {
    51→  fullName: string
    52→  email: string

    54→  course: string
    55→  academicYear: string
    56→  phone: string
    57→  bio: string
    58→  skills: string[]
    59→}
    60→
    61→interface NotifPrefs {
    62→  emailNotifs: boolean
    63→  smsNotifs: boolean
    64→  pushNotifs: boolean
    65→  inAppNotifs: boolean
    66→  reminderDays: number[]

    70→// ==================== COMPONENT ====================
    71→
    72→export function SettingsPage({ userId, userEmail }: { userId: string; userEmail?: string }) {
    73→  const { theme, setTheme } = useTheme()
    74→
    75→  // Profile state
    76→  const [profile, setProfile] = useState<ProfileData>({
    77→    fullName: '',
    78→    email: '',
    79→    college: '',
    80→    course: '',
    81→    academicYear: '',
    82→    phone: '',
    83→    bio: '',
    84→    skills: [],
    85→  })
    86→
    87→  // Notification prefs state
    88→  const [notifPrefs, setNotifPrefs] = useState<NotifPrefs>({
    89→    emailNotifs: true,
    90→    smsNotifs: false,
    91→    pushNotifs: true,
    92→    inAppNotifs: true,
    93→    reminderDays: [7, 3, 1],
    94→    reminderHours: [6, 1],
    95→  })
    96→
    97→  // Privacy state
    98→  const [publicPortfolio, setPublicPortfolio] = useState(false)
    99→  const [dataSharing, setDataSharing] = useState({ analytics: true, ai: true, team: true })
   100→
   101→  // Security state
   102→  const [currentPassword, setCurrentPassword] = useState('')
   103→  const [newPassword, setNewPassword] = useState('')
   104→  const [confirmPassword, setConfirmPassword] = useState('')
   105→
   106→  // UI state
   107→  const [loading, setLoading] = useState(true)
   108→  const [saving, setSaving] = useState<string | null>(null)
   109→  const [error, setError] = useState<string | null>(null)
   110→  const [success, setSuccess] = useState<string | null>(null)
   111→
   112→  // Fetch profile and notification prefs
   113→  useEffect(() => {
   114→    async function fetchData() {
   115→      try {
   116→        const profileRes = await fetch(`/api/onboarding?userId=${userId}`)
   117→        const profileJson = await profileRes.json()
   118→        if (profileJson.data) {
   119→          const d = profileJson.data
   120→          setProfile({
   121→            fullName: d.fullName || d.name || '',
   122→            email: d.email || userEmail || '',
   123→            college: d.college || '',
   124→            course: d.course || '',
   125→            academicYear: d.academicYear || '',
   126→            phone: d.phone || '',
   127→            bio: d.bio || '',
   128→            skills: d.skills || [],
   129→          })
   130→          // Load notification preferences if returned by the API
   131→          if (d.notificationPrefs) {
   132→            const p = d.notificationPrefs
   133→            setNotifPrefs({
   134→              emailNotifs: p.emailNotifs ?? true,
   135→              smsNotifs: p.smsNotifs ?? false,
   136→              pushNotifs: p.pushNotifs ?? true,
   137→              inAppNotifs: p.inAppNotifs ?? true,
   138→              reminderDays: safeJSONParse(p.reminderDays, [7, 3, 1]),
   139→              reminderHours: safeJSONParse(p.reminderHours, [6, 1]),
   140→            })
   141→          }
   142→        }
   143→      } catch {
   144→        // ignore — use defaults
   145→      } finally {
   146→        setLoading(false)
   147→      }
   148→    }
   149→    fetchData()
   150→  }, [userId, userEmail])
   151→
   152→  function safeJSONParse(val: unknown, fallback: number[]): number[] {
   153→    if (Array.isArray(val)) return val
   154→    if (typeof val === 'string') {
   155→      try {
   156→        const parsed = JSON.parse(val)
   157→        return Array.isArray(parsed) ? parsed : fallback
   158→      } catch {
   159→        return fallback
   160→      }
   161→    }
   162→    return fallback
   163→  }
   164→
   165→  function clearMessages() {
   166→    setError(null)
   167→    setSuccess(null)
   168→  }
   169→
   170→  // Save profile
   171→  async function saveProfile() {
   172→    clearMessages()
   173→    setSaving('profile')
   174→    try {
   175→      const res = await fetch('/api/onboarding', {
   176→        method: 'PUT',
   177→        headers: { 'Content-Type': 'application/json' },
   178→        body: JSON.stringify({
   179→          userId,
   180→          fullName: profile.fullName,
   181→          college: profile.college,
   182→          course: profile.course,
   183→          academicYear: profile.academicYear,
   184→          phone: profile.phone,
   185→          bio: profile.bio,
   186→          skills: profile.skills,
   187→        }),
   188→      })
   189→      const json = await res.json()
   190→      if (json.error) {
   191→        setError(json.error)
   192→      } else {
   193→        setSuccess('Profile updated successfully')
   194→      }
   195→    } catch {
   196→      setError('Failed to save profile')
   197→    } finally {
   198→      setSaving(null)
   199→    }
   200→  }
   201→
   202→  // Save notifications
   203→  async function saveNotifications() {
   204→    clearMessages()
   205→    setSaving('notifications')
   206→    try {
   207→      const res = await fetch('/api/notifications', {
   208→        method: 'PUT',
   209→        headers: { 'Content-Type': 'application/json' },
   210→        body: JSON.stringify({
   211→          userId,
   212→          preferences: {
   213→            emailNotifs: notifPrefs.emailNotifs,
   214→            smsNotifs: notifPrefs.smsNotifs,
   215→            pushNotifs: notifPrefs.pushNotifs,
   216→            inAppNotifs: notifPrefs.inAppNotifs,
   217→            reminderDays: JSON.stringify(notifPrefs.reminderDays),
   218→            reminderHours: JSON.stringify(notifPrefs.reminderHours),
   219→          },
   220→        }),
   221→      })
   222→      const json = await res.json()
   223→      if (json.error) {
   224→        setError(json.error)
   225→      } else {
   226→        setSuccess('Notification preferences saved')
   227→      }
   228→    } catch {
   229→      setError('Failed to save preferences')
   230→    } finally {
   231→      setSaving(null)
   232→    }
   233→  }
   234→
   235→  // Change password
   236→  async function changePassword() {
   237→    clearMessages()
   238→    if (newPassword !== confirmPassword) {
   239→      setError('Passwords do not match')
   240→      return
   241→    }
   242→    if (newPassword.length < 8) {
   243→      setError('Password must be at least 8 characters')
   244→      return
   245→    }
   246→    setSaving('security')
   247→    try {
   248→      const res = await fetch('/api/auth/change-password', {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({
   252→          currentPassword,
   253→          newPassword,
   254→        }),
   255→      })
   256→      const json = await res.json()
   257→      if (json.error) {
   258→        setError(json.error)
   259→      } else {
   260→        setSuccess('Password changed successfully')
   261→        setCurrentPassword('')
   262→        setNewPassword('')
   263→        setConfirmPassword('')
   264→      }
   265→    } catch {
   266→      setError('Failed to change password')
   267→    } finally {
   268→      setSaving(null)
   269→    }
   270→  }
   271→
   272→  // ==================== LOADING STATE ====================
   273→
   274→  if (loading) {
   275→    return (
   276→      <div className="min-h-screen bg-background p-4 md:p-6 space-y-6">
   277→        <Skeleton className="h-8 w-48" />
   278→        <Skeleton className="h-10 w-96" />
   279→        <div className="space-y-4">
   280→          <Skeleton className="h-64 rounded-xl" />
   281→          <Skeleton className="h-64 rounded-xl" />
   282→        </div>
   283→      </div>
   284→    )
   285→  }
   286→
   287→  // ==================== RENDER ====================
   288→
   289→  return (
   290→    <div className="min-h-screen bg-background p-4 md:p-6 space-y-6">
   291→      {/* Header */}
   292→      <div>
   293→        <h1 className="text-2xl font-bold flex items-center gap-2">
   294→          <SettingsIcon className="h-6 w-6 text-primary" />
   295→          Settings
   296→        </h1>
   297→        <p className="text-sm text-muted-foreground mt-1">
   298→          Manage your account and preferences
   299→        </p>
   300→      </div>
   301→
   302→      {/* Success / Error Messages */}
   303→      {success && (
   304→        <div className="p-3 rounded-lg bg-emerald-500/10 text-emerald-700 dark:text-emerald-400 text-sm flex items-center gap-2">
   305→          <Save className="h-4 w-4" /> {success}
   306→        </div>
   307→      )}
   308→      {error && (
   309→        <div className="p-3 rounded-lg bg-red-500/10 text-red-700 dark:text-red-400 text-sm flex items-center gap-2">
   310→          <AlertTriangle className="h-4 w-4" /> {error}
   311→        </div>
   312→      )}
   313→
   314→      {/* Tabs */}
   315→      <Tabs defaultValue="profile" className="space-y-4">
   316→        <TabsList className="w-full sm:w-auto flex flex-wrap gap-1">
   317→          <TabsTrigger value="profile" className="gap-1.5">
   318→            <User className="h-3.5 w-3.5" /> <span className="hidden sm:inline">Profile</span>
   319→          </TabsTrigger>

   332→        </TabsList>
   333→
   334→        {/* ==================== PROFILE TAB ==================== */}
   335→        <TabsContent value="profile">

   339→              <CardDescription>Update your personal details</CardDescription>

   342→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

   344→                  <Label htmlFor="fullName">Full Name</Label>
   345→                  <Input
   346→                    id="fullName"
   347→                    value={profile.fullName}
   348→                    onChange={(e) => setProfile({ ...profile, fullName: e.target.value })}
   349→                    placeholder="Your full name"
   350→                  />
   351→                </div>
   352→                <div className="space-y-2">
   353→                  <Label htmlFor="email">Email</Label>
   354→                  <Input
   355→                    id="email"
   356→                    value={profile.email}
   357→                    readOnly
   358→                    className="bg-muted opacity-70"
   359→                    placeholder="your@email.com"
   360→                  />
   361→                  <p className="text-xs text-muted-foreground">Email cannot be changed</p>
   362→                </div>
   363→                <div className="space-y-2">
   364→                  <Label htmlFor="college">College</Label>
   365→                  <Input
   366→                    id="college"
   367→                    value={profile.college}
   368→                    onChange={(e) => setProfile({ ...profile, college: e.target.value })}
   369→                    placeholder="Your college/university"
   370→                  />
   371→                </div>
   372→                <div className="space-y-2">
   373→                  <Label htmlFor="course">Course</Label>
   374→                  <Input
   375→                    id="course"
   376→                    value={profile.course}
   377→                    onChange={(e) => setProfile({ ...profile, course: e.target.value })}
   378→                    placeholder="Your course of study"
   379→                  />

   383→                  <Select
   384→                    value={profile.academicYear}
   385→                    onValueChange={(v) => setProfile({ ...profile, academicYear: v })}
   386→                  >
   387→                    <SelectTrigger id="academicYear">

   422→                  Save Profile

   440→                  <div>
   441→                    <p className="text-sm font-medium">Email Notifications</p>

   443→                  </div>
   444→                  <Switch
   445→                    checked={notifPrefs.emailNotifs}
   446→                    onCheckedChange={(c) => setNotifPrefs({ ...notifPrefs, emailNotifs: c })}
   447→                  />
   448→                </div>

   450→                <div className="flex items-center justify-between">

   458→                  />

   472→                <div className="flex items-center justify-between">

   480→                  />

   486→              {/* Reminder Configuration */}

   512→                </div>

   514→
   515→              <div className="space-y-3">
   516→                <div>

   521→                  {[6, 1].map((hour) => (
   522→                    <div key={hour} className="flex items-center gap-2">
   523→                      <Checkbox
   524→                        id={`hour-${hour}`}
   525→                        checked={notifPrefs.reminderHours.includes(hour)}
   526→                        onCheckedChange={(checked) => {
   527→                          setNotifPrefs({
   528→                            ...notifPrefs,
   529→                            reminderHours: checked
   530→                              ? [...notifPrefs.reminderHours, hour]
   531→                              : notifPrefs.reminderHours.filter((h) => h !== hour),
   532→                          })

   551→        </TabsContent>
   552→
   553→        {/* ==================== SECURITY TAB ==================== */}
   554→        <TabsContent value="security">
