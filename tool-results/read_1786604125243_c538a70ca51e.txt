   120→function LoginForm({ onSuccess, onSwitch }: { onSuccess: AuthPagesProps['onAuthSuccess']; onSwitch: (v: AuthView) => void }) {
   121→  const [showPassword, setShowPassword] = useState(false)
   122→  const [serverError, setServerError] = useState('')
   123→  const [loading, setLoading] = useState(false)
   124→
   125→  const form = useForm<z.infer<typeof loginSchema>>({
   126→    resolver: zodResolver(loginSchema),
   127→    defaultValues: { email: '', password: '' },
   128→  })
   129→
   130→  async function onSubmit(values: z.infer<typeof loginSchema>) {
   131→    setServerError('')
   132→    setLoading(true)
   133→    try {
   134→      const result = await signIn('credentials', {
   135→        email: values.email,
   136→        password: values.password,
   137→        redirect: false,
   138→      })
   139→
   140→      if (result?.error) {
   141→        setServerError('Invalid email or password. Please try again.')
   142→        return
   143→      }
   144→
   145→      const sessionRes = await fetch('/api/auth/session')
   146→      const session = await sessionRes.json()
   147→
   148→      if (session?.user) {
   149→        const userId = (session.user as Record<string, string>).id
   150→        if (userId) {
   151→          onSuccess({
   152→            id: userId,
   153→            email: session.user.email,
   154→            name: session.user.name || '',
   155→            role: (session.user as Record<string, string>).role || 'member',
   156→          })
   157→        } else {
   158→          const profileRes = await fetch(`/api/auth/register?email=${encodeURIComponent(values.email)}`)
   159→          const profileData = await profileRes.json()
   160→          if (profileData.data) {
   161→            onSuccess({
   162→              id: profileData.data.id,
   163→              email: profileData.data.email,
   164→              name: profileData.data.name || '',
   165→              role: profileData.data.role || 'member',
   166→            })
   167→          }
   168→        }
   169→      }
   170→    } catch {
   171→      setServerError('An unexpected error occurred. Please try again.')
   172→    } finally {
   173→      setLoading(false)
   174→    }
   175→  }
   176→
   177→  return (
   178→    <div className="space-y-5">
   179→      <div className="space-y-1.5">
   180→        <h2 className="text-2xl font-bold tracking-tight">Welcome back</h2>
   181→        <p className="text-sm text-muted-foreground">Sign in to continue to Project OS AI</p>
   182→      </div>
   183→
   184→      {serverError && (
   185→        <div className="rounded-lg border border-destructive/50 bg-destructive/10 px-4 py-3 text-sm text-destructive">
   186→          {serverError}
   187→        </div>
   188→      )}
   189→
   190→      <Form {...form}>
   191→        <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   192→          <FormField
   193→            control={form.control}
   194→            name="email"
   195→            render={({ field }) => (
   196→              <FormItem>
   197→                <FormLabel>Email</FormLabel>
   198→                <FormControl>
   199→                  <div className="relative">
   200→                    <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   201→                    <Input placeholder="you@example.com" className="pl-10" {...field} />
   202→                  </div>
   203→                </FormControl>
   204→                <FormMessage />
   205→              </FormItem>
   206→            )}
   207→          />
   208→          <FormField
   209→            control={form.control}
   210→            name="password"
   211→            render={({ field }) => (
   212→              <FormItem>
   213→                <div className="flex items-center justify-between">
   214→                  <FormLabel>Password</FormLabel>
   215→                  <Button type="button" variant="link" className="px-0 h-auto text-xs text-muted-foreground hover:text-primary" onClick={() => onSwitch('forgot-password')}>

   223→                    <button

   225→                      onClick={() => setShowPassword(!showPassword)}

   260→function SignupForm({ onSuccess, onSwitch }: { onSuccess: AuthPagesProps['onAuthSuccess']; onSwitch: (v: AuthView) => void }) {
   261→  const [showPassword, setShowPassword] = useState(false)
   262→  const [showConfirmPassword, setShowConfirmPassword] = useState(false)
   263→  const [serverError, setServerError] = useState('')
   264→  const [loading, setLoading] = useState(false)
   265→
   266→  const form = useForm<z.infer<typeof signupSchema>>({
   267→    resolver: zodResolver(signupSchema),
   268→    defaultValues: { name: '', email: '', password: '', confirmPassword: '' },
   269→  })
   270→
   271→  async function onSubmit(values: z.infer<typeof signupSchema>) {
   272→    setServerError('')
   273→    setLoading(true)
   274→    try {
   275→      const res = await fetch('/api/auth/register', {
   276→        method: 'POST',
   277→        headers: { 'Content-Type': 'application/json' },
   278→        body: JSON.stringify({ name: values.name, email: values.email, password: values.password }),
   279→      })
   280→
   281→      const regData = await res.json()
   282→
   283→      if (!res.ok) {
   284→        setServerError(regData.error || 'Registration failed. Please try again.')
   285→        return
   286→      }
   287→
   288→      const registeredUser = regData.data
   289→
   290→      // Sign in to establish session cookie
   291→      const result = await signIn('credentials', {
   292→        email: values.email,
   293→        password: values.password,
   294→        redirect: false,
   295→      })
   296→
   297→      if (!result?.ok) {
   298→        setServerError('Registration succeeded but login failed. Please try signing in manually.')
   299→        return
   300→      }
   301→
   302→      // Small delay to ensure session cookie is set
   303→      await new Promise((r) => setTimeout(r, 300))
   304→
   305→      const sessionRes = await fetch('/api/auth/session')
   306→      const session = await sessionRes.json()
   307→      const sessionUser = session?.user as Record<string, string> | undefined
   308→
   309→      if (sessionUser?.id) {
   310→        onSuccess({
   311→          id: sessionUser.id,
   312→          email: sessionUser.email || values.email,
   313→          name: sessionUser.name || values.name,
   314→          role: sessionUser.role || 'member',
   315→        })
   316→      } else if (registeredUser) {
   317→        // Fallback: use registered user data if session fetch failed
   318→        onSuccess({
   319→          id: registeredUser.id,
   320→          email: registeredUser.email,
   321→          name: registeredUser.name || values.name,
   322→          role: registeredUser.role || 'member',
   323→        })
   324→      }
   325→    } catch {
   326→      setServerError('An unexpected error occurred. Please try again.')
   327→    } finally {
   328→      setLoading(false)
   329→    }
   330→  }
   331→
   332→  return (
   333→    <div className="space-y-5">
   334→      <div className="space-y-1.5">
   335→        <h2 className="text-2xl font-bold tracking-tight">Create your account</h2>

   337→      </div>
   338→
   339→      {serverError && (
   340→        <div className="rounded-lg border border-destructive/50 bg-destructive/10 px-4 py-3 text-sm text-destructive">
   341→          {serverError}
   342→        </div>
   343→      )}
   344→
   345→      <Form {...form}>
   346→        <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   347→          <FormField
   348→            control={form.control}
   349→            name="name"
   350→            render={({ field }) => (
   351→              <FormItem>
   352→                <FormLabel>Full Name</FormLabel>
   353→                <FormControl>
   354→                  <div className="relative">
   355→                    <User className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
