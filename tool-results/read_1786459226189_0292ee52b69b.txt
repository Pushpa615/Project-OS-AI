   142→   142→function parseJSONField(val: string | null | undefined): string[] {
   143→   143→  if (!val) return []
   144→   144→  try {
   145→   145→    const parsed = JSON.parse(val)
   146→   146→    return Array.isArray(parsed) ? parsed : []
   147→   147→  } catch {
   148→   148→    return []
   149→   149→  }
   150→   150→}
   151→   151→
   152→   152→function parseJSONObj(val: string | null | undefined): Record<string, unknown> {
   153→   153→  if (!val) return {}
   154→   154→  try {
   155→   155→    return JSON.parse(val)
   156→   156→  } catch {
   157→   157→    return {}
   158→   158→  }
   159→   159→}
