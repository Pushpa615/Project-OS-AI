    69→function formatDate(dateStr: string | null): string {
    70→  if (!dateStr) return 'No deadline'
    71→  try {
    72→    return new Date(dateStr).toLocaleDateString('en-US', {
    73→      month: 'short',
    74→      day: 'numeric',
    75→      year: 'numeric',
    76→    })
    77→  } catch {
    78→    return 'Invalid date'
    79→  }
    80→}
    81→
    82→function formatRelativeTime(dateStr: string): string {
    83→  try {
    84→    const date = new Date(dateStr)
    85→    const now = new Date()
    86→    const diffMs = now.getTime() - date.getTime()
    87→    const diffMins = Math.floor(diffMs / 60000)
    88→    if (diffMins < 1) return 'Just now'
    89→    if (diffMins < 60) return `${diffMins}m ago`
    90→    const diffHours = Math.floor(diffMins / 60)
    91→    if (diffHours < 24) return `${diffHours}h ago`
    92→    const diffDays = Math.floor(diffHours / 24)
    93→    if (diffDays < 7) return `${diffDays}d ago`
    94→    return formatDate(dateStr)
    95→  } catch {
    96→    return ''
    97→  }
    98→}

   100→function getStatusColor(status: string) {
   101→  switch (status) {
   102→    case 'in_progress':
   103→      return 'bg-amber-500/10 text-amber-700 dark:text-amber-400 border-amber-500/20'
   104→    case 'completed':
   105→    case 'verified_completed':
   106→      return 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-400 border-emerald-500/20'
   107→    case 'not_started':
   108→      return 'bg-muted text-muted-foreground border-border'
   109→    case 'on_hold':
   110→      return 'bg-orange-500/10 text-orange-700 dark:text-orange-400 border-orange-500/20'
   111→    case 'overdue':
   112→      return 'bg-red-500/10 text-red-700 dark:text-red-400 border-red-500/20'
   113→    default:
   114→      return 'bg-muted text-muted-foreground border-border'
   115→  }
   116→}
